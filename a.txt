Step 2: Modify proc.c — Initialize priority
Open:
nano proc.c

Inside allocproc(), find:

Step 3: Modify proc.c — Inherit priority in fork()
In the same file, find:

Add immediately after:

Step 4: Modify proc.c — Replace the scheduler

void
scheduler(void)
{
  struct proc *p, *best;
  struct cpu *c = mycpu();
  c->proc = 0;

  for(;;){
    sti();
    acquire(&ptable.lock);

    best = 0;
    for(p = ptable.proc; p < &ptable.proc[NPROC]; p++){
      if(p->state == RUNNABLE){
        if(best == 0 || p->priority > best->priority)
          best = p;
      }
    }

    if(best != 0){
      c->proc = best;
      switchuvm(best);
      best->state = RUNNING;
      swtch(&(c->scheduler), best->context);
      switchkvm();
      c->proc = 0;
    }

    release(&ptable.lock);
  }
}

Step 7: Modify sysproc.c
Open:
nano sysproc.c

Add this function:
int
sys_party(void)
{
  int priority;

  if(argint(0, &priority) < 0)
    return -1;

  if(priority < MIN_PRIORITY || priority > MAX_PRIORITY)
    return -1;

  myproc()->priority = priority;
  return 0;
}

Step 10: Create the test program
Create a new file:
nano priority.c

#include "types.h"
#include "stat.h"
#include "user.h"

int
main(void)
{
  int i, pid;
  int priorities[] = {1, 2, 3, 4};

  for(i = 0; i < 4; i++){
    pid = fork();

    if(pid == 0){
      int j;
      int pr = priorities[i];

      if(party(pr) < 0){
        printf(1, "Failed to set priority\n");
        exit();
      }

      printf(1, "PID %d Priority %d started\n", getpid(), pr);

      for(j = 0; j < 5000000; j++)
        ;

      printf(1, "PID %d Priority %d finished\n", getpid(), pr);
      exit();
    }
  }

  for(i = 0; i < 4; i++)
    wait();

  exit();
}

Step 13: Create the patch file
Return to the xv6 directory if necessary:
cd ~/OS/priority/xv6-public

Check the changed files:
git status

Stage the implementation files:
git add Makefile proc.h proc.c syscall.h syscall.c sysproc.c user.h usys.S priority.c

Generate the patch:
git diff --cached > priority.patch

Verify it:
